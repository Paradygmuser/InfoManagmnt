CREATE VIEW NPSAD AS
SELECT
    E.nom,E.prenom,
    (SELECT E.salaire*1.25 GROUP BY E.nss DESC LIMIT 10) AS salaire_dollar,
    D.nom AS nomdepartement,
    (SELECT TIMESTAMPDIFF(year,E.datenaissance,now())) AS age
FROM Employe E
JOIN Departement D on D.numerodept = E.deptnr;
